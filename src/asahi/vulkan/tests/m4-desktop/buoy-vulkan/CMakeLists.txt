cmake_minimum_required(VERSION 3.16)
project(buoy_vulkan LANGUAGES CXX)
find_package(Vulkan REQUIRED)
find_package(SDL2 REQUIRED CONFIG)
add_executable(buoy-vulkan main.cpp)
target_compile_features(buoy-vulkan PRIVATE cxx_std_17)
target_compile_options(buoy-vulkan PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
if(APPLE AND TARGET SDL2::SDL2-static)
  target_link_libraries(buoy-vulkan PRIVATE SDL2::SDL2-static Vulkan::Vulkan)
else()
  target_link_libraries(buoy-vulkan PRIVATE SDL2::SDL2 Vulkan::Vulkan)
endif()
